{{- /* Parses the page bundle's .ipynb (nbformat 4) into a flat list of cells. Called through partialCached, so the post page, the search index and the LLM-copy payload share one parse. Returns: dict kernel display name of the kernel ("Python 3 (ipykernel)") lang notebook default language (python, cpp, ...) file the .ipynb resource cells slice of dict: i 0-based index (anchor #cell-i) type markdown | code | raw role Socratech journal role from cell tags (exercise | attempt | verdict | reflection | aside | agent | code) or "" math true for math cells (checkered paper) lang highlight language for code cells magic leading %%magic line, shown apart from the highlighted body body source without the magic line src raw source, untouched (what gets copied) html rendered markdown (markdown cells only) count execution count or nil outputs slice of dict {kind, value} with kind in text | stderr | error | html | md | img | svg outtext plain-text rendition of outputs, for copying and search */ -}} {{- $page := . -}} {{- $res := .Resources.GetMatch "*.ipynb" -}} {{- $nb := $res.Content | transform.Unmarshal -}} {{- $meta := $nb.metadata | default dict -}} {{- $li := $meta.language_info | default dict -}} {{- $ks := $meta.kernelspec | default dict -}} {{- /* Language aliases → Chroma lexer names. */ -}} {{- $alias := dict "python" "python" "python3" "python" "ipython" "python" "ipython3" "python" "c++" "cpp" "cpp" "cpp" "c++11" "cpp" "c++14" "cpp" "c++17" "cpp" "c++20" "cpp" "c++23" "cpp" "c" "c" "bash" "bash" "sh" "bash" "shell" "bash" "script" "bash" "latex" "latex" "tex" "latex" "html" "html" "javascript" "js" "js" "js" "markdown" "markdown" "julia" "julia" "rust" "rust" "haskell" "haskell" "lean" "lean" -}} {{- $nblang := lower (string ($li.name | default $ks.language | default "python")) -}} {{- $nblang = index $alias $nblang | default $nblang -}} {{- $extlang := dict "cpp" "cpp" "cc" "cpp" "cxx" "cpp" "hpp" "cpp" "h" "c" "c" "c" "py" "python" "sh" "bash" "tex" "latex" "rs" "rust" "hs" "haskell" "lean" "lean" -}} {{- $mathRE := `(?s)^\s*((\$\$.+?\$\$|\\\[.+?\\\])\s*)+$` -}} {{- $ansiRE := "\x1b\\[[0-9;?]*[A-Za-z]" -}} {{- /* Socratech journal exports tag each cell with its role. */ -}} {{- $roles := slice "exercise" "attempt" "verdict" "reflection" "aside" "agent" "code" -}} {{- $cells := slice -}} {{- range $i, $c := $nb.cells -}} {{- $src := partial "nb/join.html" $c.source -}} {{- $cm := $c.metadata | default dict -}} {{- $tags := $cm.tags | default slice -}} {{- $role := "" -}} {{- range $tags -}}{{- if and (not $role) (in $roles .) -}}{{- $role = . -}}{{- end -}}{{- end -}} {{- $cell := dict "i" $i "type" $c.cell_type "role" $role "src" $src "body" $src "magic" "" "math" false "lang" "" "html" "" "count" nil "outputs" slice "outtext" "" -}} {{- if eq $c.cell_type "markdown" -}} {{- /* Math cell: only display math, the "math" tag, or a leading comment. */ -}} {{- $isMath := or (findRE $mathRE $src 1) (in $tags "math") (hasPrefix (trim $src " \n") "") -}} {{- /* Pasted images live in cell attachments; inline them as data URIs. */ -}} {{- $md := $src -}} {{- range $name, $att := $c.attachments -}} {{- range $mime, $data := $att -}} {{- $md = replace $md (printf "attachment:%s" $name) (printf "data:%s;base64,%s" $mime (replace (partial "nb/join.html" $data) "\n" "")) -}} {{- end -}} {{- end -}} {{- $cell = merge $cell (dict "math" (not (not $isMath)) "html" ($page.RenderString (dict "display" "block") $md)) -}} {{- else if eq $c.cell_type "code" -}} {{- $lang := $nblang -}} {{- with $cm.language -}}{{- $lang = index $alias (lower .) | default (lower .) -}}{{- end -}} {{- with $cm.vscode -}}{{- with .languageId -}}{{- $lang = index $alias (lower .) | default (lower .) -}}{{- end -}}{{- end -}} {{- $magic := "" -}} {{- $body := $src -}} {{- $first := index (split $src "\n") 0 -}} {{- if hasPrefix $first "%%" -}} {{- $name := lower (replaceRE `^%%(\S+).*$` "$1" $first) -}} {{- $magic = $first -}} {{- $body = strings.TrimPrefix "\n" (strings.TrimPrefix $first $src) -}} {{- if eq $name "writefile" -}} {{- $ext := lower (replaceRE `^.*\.(\w+)\s*$` "$1" $first) -}} {{- with index $extlang $ext -}}{{- $lang = . -}}{{- end -}} {{- else -}} {{- with index $alias $name -}}{{- $lang = . -}}{{- end -}} {{- end -}} {{- end -}} {{- $outs := slice -}} {{- $txt := slice -}} {{- range $c.outputs -}} {{- $ot := .output_type -}} {{- if eq $ot "stream" -}} {{- $v := replaceRE $ansiRE "" (partial "nb/join.html" .text) -}} {{- $outs = $outs | append (dict "kind" (cond (eq .name "stderr") "stderr" "text") "value" $v) -}} {{- $txt = $txt | append $v -}} {{- else if eq $ot "error" -}} {{- $v := replaceRE $ansiRE "" (delimit (.traceback | default slice) "\n") -}} {{- if not $v -}}{{- $v = printf "%s: %s" .ename .evalue -}}{{- end -}} {{- $outs = $outs | append (dict "kind" "error" "value" $v) -}} {{- $txt = $txt | append $v -}} {{- else if in (slice "execute_result" "display_data") $ot -}} {{- $d := .data | default dict -}} {{- $plain := partial "nb/join.html" (index $d "text/plain") -}} {{- $png := index $d "image/png" -}} {{- $jpg := index $d "image/jpeg" -}} {{- $svg := index $d "image/svg+xml" -}} {{- $html := index $d "text/html" -}} {{- $mdo := or (index $d "text/markdown") (index $d "text/latex") -}} {{- if or $png $jpg -}} {{- $mime := cond (not (not $png)) "image/png" "image/jpeg" -}} {{- $outs = $outs | append (dict "kind" "img" "value" (printf "data:%s;base64,%s" $mime (replace (partial "nb/join.html" (or $png $jpg)) "\n" ""))) -}} {{- $txt = $txt | append "[image output]" -}} {{- else if $svg -}} {{- $outs = $outs | append (dict "kind" "svg" "value" (partial "nb/join.html" $svg)) -}} {{- $txt = $txt | append "[svg output]" -}} {{- else if $html -}} {{- $outs = $outs | append (dict "kind" "html" "value" (partial "nb/join.html" $html)) -}} {{- $txt = $txt | append ($plain | default "[html output]") -}} {{- else if $mdo -}} {{- $v := partial "nb/join.html" $mdo -}} {{- $outs = $outs | append (dict "kind" "md" "value" ($page.RenderString (dict "display" "block") $v)) -}} {{- $txt = $txt | append $v -}} {{- else -}} {{- $v := replaceRE $ansiRE "" $plain -}} {{- $outs = $outs | append (dict "kind" "text" "value" $v) -}} {{- $txt = $txt | append $v -}} {{- end -}} {{- end -}} {{- end -}} {{- $cell = merge $cell (dict "lang" $lang "magic" $magic "body" $body "count" $c.execution_count "outputs" $outs "outtext" (delimit $txt "\n")) -}} {{- end -}} {{- $cells = $cells | append $cell -}} {{- end -}} {{- return dict "kernel" ($ks.display_name | default (title $nblang)) "lang" $nblang "file" $res "cells" $cells -}}
